1Use font-face declaration Fonts. 
@font-face {
  font-family:"monoflow";
  font-weight:normal;
  font-style:italic;
  src:url("./monoflow-italic.woff2")format("woff2");
}
@font-face {
  font-family:'monoflow';
  font-weight:normal;
  font-style:normal;
  src:url('./monoflow.woff2')format("woff2");
}
@font-face{
  font-family:"Monoflow";
  src:url("monoflow.eot")format("embedded-opentype"),
    url("monoflow.eot?#iefix")format("embedded-opentype"),
    url("monoflow.woff")format("woff"),
    url("monoflow.woff2")format("woff2"),
    url("monoflow.ttf")format("truetype"),
    url("monoflow.svg#Monoflow")format("svg");
  font-weight:normal;
  font-style:normal;
  font-display:swap;
}
