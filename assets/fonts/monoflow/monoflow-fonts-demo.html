	font-family:'Monoflow';

			format("embedded-opentype"),
	 url("./monoflow.eot?#iefix")
			format("embedded-opentype"),
	 url("./monoflow.woff")format("woff"),
	 url("./monoflow.woff2")format("woff2"),
	 url("./monoflow.ttf")format("truetype"),
	 url("./monoflow.svg#Monoflow")
			format("svg");
	font-weight:normal;
	font-style:normal;
	font-display:swap;
}
